{{- /* Start button — a green play button labeled "start learning" that links to the next chapter in the course (sidebar order: topic_weight, subtopic_weight, weight). On a course intro page it links to the first chapter. Renders nothing when there is no next chapter. Usage: {{< start >}} */ -}} {{- $next := false -}} {{- $parts := slice -}} {{- with .Page.File }}{{ $parts = split (strings.TrimSuffix "/" .Dir) "/" }}{{ end -}} {{- if ge (len $parts) 2 -}} {{- $topic := index $parts 1 -}} {{- $isIntro := eq .Page.Kind "section" -}} {{- $tw := 0 }}{{ $sw := 0 }}{{ $w := 0 -}} {{- if not $isIntro -}} {{- $tw = cond (isset .Page.Params "topic_weight") (int .Page.Params.topic_weight) 99 -}} {{- $sw = cond (isset .Page.Params "subtopic_weight") (int .Page.Params.subtopic_weight) 99 -}} {{- $w = cond (isset .Page.Params "weight") (int .Page.Params.weight) 99 -}} {{- end -}} {{- $keys := slice -}} {{- $pageMap := dict -}} {{- range where site.RegularPages "Section" "courses" -}} {{- if .File -}} {{- $p := split (strings.TrimSuffix "/" .File.Dir) "/" -}} {{- if and (ge (len $p) 3) (eq (index $p 1) $topic) -}} {{- $ctw := cond (isset .Params "topic_weight") (int .Params.topic_weight) 99 -}} {{- $csw := cond (isset .Params "subtopic_weight") (int .Params.subtopic_weight) 99 -}} {{- $cw := cond (isset .Params "weight") (int .Params.weight) 99 -}} {{- $key := printf "%04d-%04d-%04d-%s" $ctw $csw $cw .File.Path -}} {{- $keys = $keys | append $key -}} {{- $pageMap = merge $pageMap (dict $key .) -}} {{- end -}} {{- end -}} {{- end -}} {{- $keys = sort $keys -}} {{- if $isIntro -}} {{- with index $pageMap (index $keys 0) }}{{ $next = . }}{{ end -}} {{- else -}} {{- $cur := printf "%04d-%04d-%04d-%s" $tw $sw $w .Page.File.Path -}} {{- $found := false -}} {{- range $keys -}} {{- if $found }}{{ $next = index $pageMap . }}{{ break }}{{ end -}} {{- if eq . $cur }}{{ $found = true }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- with $next -}} start learning {{- end -}}
